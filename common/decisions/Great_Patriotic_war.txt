KOA_Bio_Agents = {
	KOA_Anti_Radiation_Extremophilic_Bacterias = {
		# allows to make nuclear contamination less hazardous in a chosen state
		state_target = yes
		on_map_mode = map_only
		target_array = global.states
		target_trigger = {
			FROM = {
				is_controlled_by_ROOT_or_ally = yes
				OR = {
					has_dynamic_modifier = Radiation_heavy
					has_dynamic_modifier = Radiation_medium
					has_dynamic_modifier = Radiation_light
				}
			}
		}
		cost = 5
		days_remove = 180
		complete_effect = {
		}
		modifier = {
			civilian_factory_use = 5
		}
		remove_effect = {
			FROM = {
				if = {
					limit = {
						has_dynamic_modifier = Radiation_heavy
					}
					remove_dynamic_modifier = {
						modifier = Radiation_heavy
					}
					add_dynamic_modifier = {
						modifier = Radiation_medium
					}
				}
				else_if = {
					limit = {
						has_dynamic_modifier = Radiation_medium
					}
					remove_dynamic_modifier = {
						modifier = Radiation_medium
					}
					add_dynamic_modifier = {
						modifier = Radiation_light
					}
				}
				else_if = {
					limit = {
						has_dynamic_modifier = Radiation_light
					}
					remove_dynamic_modifier = {
						modifier = Radiation_light
					}
					if = {
						limit = {
							KOA = {
								has_stability < 0.49
							}
						}
						KOA = {
							add_stability = 0.01
						}
					}
				}
			}
		}
		ai_will_do = {
			base = 1
		}
	}
